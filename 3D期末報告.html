playPromise.then(() => {

iconPlay.classList.add('hidden');

iconPause.classList.remove('hidden');

}).catch(error => {

console.warn('影片播放被中斷:', error);

});

}

}

}

 

function toggleVideo() {

if (video.paused) {

const playPromise = video.play();

if (playPromise !== undefined) {

playPromise.then(() => {

// 播放成功

iconPlay.classList.add('hidden');

iconPause.classList.remove('hidden');

}).catch(error => {

// 捕捉 AbortError (例如使用者快速連點暫停)

console.warn('影片播放被中斷:', error);

});

}

} else {

video.pause();

iconPause.classList.add('hidden');

iconPlay.classList.remove('hidden');

}

}

async function fetchPokemon(name) {

if (!name) return;

 

resultPanel.style.opacity = '0';

setTimeout(() => resultPanel.classList.add('hidden'), 300);

 

statusMessage.classList.remove('hidden');

loader.classList.remove('hidden');

statusText.innerText = `[ 系統檢索中 ] 目標：${name.toUpperCase()} ...`;

 

try {

const response = await fetch(`https://pokeapi.co/api/v2/pokemon/${name.toLowerCase()}`);

if (!response.ok) throw new Error('資料庫無法匹配該名稱，請確認拼寫是否正確。');

 

const data = await response.json();

renderPokemon(data);

 

statusMessage.classList.add('hidden');

resultPanel.classList.remove('hidden');

setTimeout(() => resultPanel.style.opacity = '1', 50);

 

} catch (error) {

loader.classList.add('hidden');

statusText.innerHTML = `<span class="text-red-500 font-bold">錯誤：</span> ${error.message}`;

}

}

const span = document.createElement('span');

span.className = `type-badge type-${type} shadow-lg`;

span.innerText = typeTranslations[type] || type;

typesContainer.appendChild(span);

});

 

updateChart(data.stats);

calculateWeaknesses(currentTypes);

}

 

function updateChart(stats) {

const labels = ['HP (體力)', 'ATK (攻擊)', 'DEF (防禦)', 'SPA (特攻)', 'SPD (特防)', 'SPE (速度)'];

const values = stats.map(s => s.base_stat);

 

if (statsChart) statsChart.destroy();

 

const ctx = document.getElementById('statsChart').getContext('2d');

 

// 根據能力值總和改變顏色

const totalStats = values.reduce((a, b) => a + b, 0);

const chartColor = totalStats > 500 ? 'rgba(59, 130, 246, 0.5)' : 'rgba(239, 68, 68, 0.4)';

const borderColor = totalStats > 500 ? '#3b82f6' : '#ef4444';

 

statsChart = new Chart(ctx, {

type: 'radar',

data: {

labels: labels,

datasets: [{

label: '種族值',

data: values,

fill: true,

backgroundColor: chartColor,

borderColor: borderColor,

pointBackgroundColor: '#fff',

pointBorderColor: borderColor,

borderWidth: 2,

pointHoverRadius: 6

}]

},

options: {

responsive: true,

maintainAspectRatio: false,

scales: {

r: {

angleLines: { color: 'rgba(255,255,255,0.15)' },

grid: { color: 'rgba(255,255,255,0.15)' },

pointLabels: { color: '#cbd5e1', font: { size: 13, weight: 'bold', family: "'Noto Sans TC', sans-serif" } },

ticks: { display: false, stepSize: 40 },

suggestedMin: 0,

suggestedMax: 160

}

},

plugins: { legend: { display: false } }

}

});

}

 

function calculateWeaknesses(types) {

const container = document.getElementById('weaknessList');

container.innerHTML = '';

 

let multipliers = {};

const allTypes = Object.keys(typeEffectiveness);

allTypes.forEach(t => multipliers[t] = 1);

 

types.forEach(pType => {

const rules = typeEffectiveness[pType];

if (rules) {

rules.weaknesses?.forEach(w => multipliers[w] *= 2);

rules.immunities?.forEach(i => multipliers[i] *= 0);

}

});

 

let hasWeakness = false;

for (const [type, mult] of Object.entries(multipliers)) {

if (mult > 1) {

hasWeakness = true;

const wrapper = document.createElement('div');

wrapper.className = 'flex flex-col items-center gap-2 bg-slate-800 p-2 rounded-lg border border-slate-700 min-w-[70px]';

 

const badge = document.createElement('span');

badge.className = `type-badge type-${type} w-full text-center`;

badge.innerText = typeTranslations[type];

 

const factor = document.createElement('span');

factor.className = `text-lg font-black ${mult >= 4 ? 'text-red-500 animate-pulse' : 'text-orange-400'}`;

factor.innerText = `${mult}x`;

 

wrapper.appendChild(badge);

wrapper.appendChild(factor);

container.appendChild(wrapper);

}

}

 

if (!hasWeakness) {

container.innerHTML = '<div class="flex items-center gap-2 text-green-400 font-bold"><svg class="w-6 h-6" fill="currentColor" viewBox="0 0 20 20"><path fill-rule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zm3.707-9.293a1 1 0 00-1.414-1.414L9 10.586 7.707 9.293a1 1 0 00-1.414 1.414l2 2a1 1 0 001.414 0l4-4z" clip-rule="evenodd"></path></svg>目標沒有明顯的防禦弱點</div>';

}

}
